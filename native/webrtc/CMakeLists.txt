cmake_minimum_required(VERSION 3.21)
project(remoteapp_webrtc LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(NO_EXAMPLES ON CACHE BOOL "" FORCE)
set(NO_TESTS ON CACHE BOOL "" FORCE)
set(NO_WEBSOCKET ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_DEPS_LIBS OFF CACHE BOOL "" FORCE)
if(APPLE)
  set(OPENSSL_USE_STATIC_LIBS ON CACHE BOOL "" FORCE)
endif()
if(NOT REMOTEAPP_LIBDATACHANNEL_SOURCE)
  message(FATAL_ERROR "Use scripts/build-webrtc.py to provide pinned libdatachannel sources")
endif()
add_subdirectory("${REMOTEAPP_LIBDATACHANNEL_SOURCE}" datachannel EXCLUDE_FROM_ALL)
add_library(remoteapp_webrtc SHARED peer.cpp)
target_link_libraries(remoteapp_webrtc PRIVATE datachannel-static)
target_compile_options(remoteapp_webrtc PRIVATE -Wall -Wextra -Werror)
set_target_properties(remoteapp_webrtc PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
add_executable(remoteapp_webrtc_test wire_test.cpp)
target_compile_options(remoteapp_webrtc_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
enable_testing()
add_test(NAME media_wire COMMAND remoteapp_webrtc_test)
add_executable(remoteapp_webrtc_receiver_test receiver_test.cpp)
target_link_libraries(remoteapp_webrtc_receiver_test PRIVATE datachannel-static)
target_compile_options(remoteapp_webrtc_receiver_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME media_repair COMMAND remoteapp_webrtc_receiver_test)
